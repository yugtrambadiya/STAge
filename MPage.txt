    Remainder dw 0

; EXPERIMENT 10 - FILE 2: DIVISION PROCEDURE MODULE
Code_here segment public 'CODE'
Assume cs:Code_here,ds:Data_here
Extrn Numerator:dword
Extrn Divisor:word
Extrn Quotient:dword
Extrn Remainder:word
Public Divide

Divide proc far
    mov ax,Numerator+2
    xor dx,dx
    div Divisor
    mov Quotient+2,ax
    mov ax,Numerator
    div Divisor
    mov Quotient,ax
    mov Remainder,dx
    ret
Divide endp

Code_here ends
End
